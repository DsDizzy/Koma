-- a profile is created automatically when someone signs up (username comes from sign-up form) create function public.handle_new_user() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.profiles (id, username) values (new.id, lower(new.raw_user_meta_data->>'username')); return new; end $$; create trigger on_auth_user_created after insert on auth.users for each row execute function public.handle_new_user(); create table public.creator_verifications ( user_id uuid primary key references public.profiles on delete cascade, pen_name text not null, titles text, declaration_signed_at timestamptz not null default now(), status text not null default 'pending' check (status in ('pending','verified','rejected')), token text unique, reviewed_at timestamptz ); -- when you set status = 'verified' in the Table Editor, a verification token is issued create function public.verif_token() returns trigger language plpgsql as $$ begin if new.status = 'verified' and new.token is null then new.token := 'KOMA-VRF-' || upper(substr(md5(random()::text || clock_timestamp()::text), 1, 12)); new.reviewed_at := now(); end if; return new; end $$; create trigger verif_token before update on public.creator_verifications for each row execute function public.verif_token(); create function public.is_verified(uid uuid) returns boolean language sql stable security definer set search_path = public as $$ select exists(select 1 from creator_verifications where user_id = uid and status = 'verified') $$; -- ---------- titles & content ---------- -- every readable title (built-in ones + creator uploads) and how many chapters it has create table public.titles (id text primary key, chapter_count int not null default 0); insert into public.titles values ('windbound', 2), ('pepper-carrot', 4); create table public.manga ( id uuid primary key default gen_random_uuid(), creator_id uuid not null references public.profiles, title text not null, description text, genres text[] not null default '{}', direction text not null default 'ltr' check (direction in ('ltr','rtl')), cover_path text, created_at timestamptz not null default now() ); create table public.chapters ( id uuid primary key default gen_random_uuid(), manga_id uuid not null references public.manga on delete cascade, number int not null, title text not null, pages jsonb not null, published_at timestamptz not null default now(), unique (manga_id, number) ); create table public.panel_scans (panel_id text primary key, tags text[] not null default '{}', text text, scanned_at timestamptz default now()); -- ---------- reading state ---------- create table public.progress ( user_id uuid references public.profiles on delete cascade, manga_id text not null, chapters_done int[] not null default '{}', current_chapter int not null default 0, position int not null default 0, updated_at timestamptz not null default now(), primary key (user_id, manga_id) ); create table public.list_items (user_id uuid references public.profiles on delete cascade, manga_id text not null, added_at timestamptz default now(), primary key (user_id, manga_id)); create function public.finished(uid uuid, mid text) returns boolean language sql stable security definer set search_path = public as $$ select coalesce((select cardinality(chapters_done) from progress where user_id = uid and manga_id = mid), 0) >= greatest(1, coalesce((select chapter_count from titles where id = mid), (select count(*)::int from chapters where manga_id::text = mid), 0)) $$; create function public.chapters_read(uid uuid, mid text) returns int language sql stable security definer set search_path = public as $$ select coalesce((select max(x) + 1 from progress p, unnest(p.chapters_done) x where p.user_id = uid and p.manga_id = mid), 0) $$; create table public.ratings ( user_id uuid references public.profiles on delete cascade, manga_id text not null, stars int not null check (stars between 1 and 5), primary key (user_id, manga_id) ); -- ---------- community ---------- create table public.threads ( id uuid primary key default gen_random_uuid(), manga_id text not null, author_id uuid references public.profiles on delete cascade, author_name text, -- author_name only for seeded demo posts kind text not null check (kind in ('Discussion','Theory','Reaction','Question')), title text not null check (char_length(title) <= 140), body text not null check (char_length(body) <= 4000), spoiler_ch int not null default 0, created_at timestamptz not null default now() ); create table public.replies ( id uuid primary key default gen_random_uuid(), thread_id uuid not null references public.threads on delete cascade, author_id uuid references public.profiles on delete cascade, author_name text, body text not null check (char_length(body) <= 2000), spoiler_ch int not null default 0, created_at timestamptz not null default now() ); create table public.news_posts ( id uuid primary key default gen_random_uuid(), kind text not null check (kind in ('headline','creator')), author_id uuid references public.profiles on delete cascade, author_name text, manga_id text, source text, url text, title text not null check (char_length(title) <= 160), body text not null check (char_length(body) <= 4000), created_at timestamptz not null default now() ); create table public.news_comments ( id uuid primary key default gen_random_uuid(), post_id uuid not null references public.news_posts on delete cascade, author_id uuid references public.profiles on delete cascade, author_name text, body text not null check (char_length(body) <= 600), created_at timestamptz not null default now() );
